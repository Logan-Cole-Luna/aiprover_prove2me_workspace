\title{The inverse of $X_{-1}(u)$ in the group $G(\mathfrak{m})$: a machine-checked proof}
\date{}
\begin{document}
\maketitle
\begin{abstract}
We present a formal verification of Corollary 3.2 (label \texttt{cor: gsimple x inverse}) from a text on the Monster Lie algebra group $G(\mathfrak{m})$. The group is defined by generators and relations, and the corollary states that $X_{-1}(u)^{-1}=X_{-1}(-u)$ for every $u\in\mathbb{C}$. In the formalization the group is the presented group on the generating symbols $\mathcal{X}$ modulo the full list of defining relations. The index set $E$ is built from an arbitrary function $c:\mathbb{N}\to\mathbb{N}$ in place of the coefficients of the modular function $J$. The identity is proved in Lean 4 with Mathlib for every such $c$, using only the relation (Re:1a).
\end{abstract}

\section{Statement}

Let $\mathbb{C}$ denote the complex numbers and $\mathbb{N}$ the natural numbers. Let $c(j)$ denote the coefficient of $q^j$ in the modular function
\[
J(q)=\sum_{j\geq -1}c(j)q^j=\frac{1}{q}+196884\,q+21493760\,q^2+864299970\,q^3+\cdots,
\]
so that $c(-1)=1$, $c(0)=0$, $c(1)=196884,\dots$.

\paragraph{Index set and constants.} The index set is
\[
E=\{(\ell,j,k)\mid j\in\mathbb{N},\ 1\leq k\leq c(j),\ 0\leq \ell<j\}.
\]
For $(\ell,j,k)\in E$ we set
\[
c_{\ell j}:=(-1)^{\ell+1}\binom{j-1}{\ell}(\ell+1)(j-\ell)\in\mathbb{C}.
\]

\paragraph{Generators.} Let $\cX$ be the set of symbols
\[
\{H_1(s),H_2(s),X_{-1}(u),Y_{-1}(u),X_{\ell,jk}(u),Y_{\ell,jk}(u)\mid s\in\Cx,\ u\in\mathbb{C},\ (\ell,j,k)\in E\},
\]
and let $F(\cX)$ be the free group on $\cX$.

\paragraph{Relations.} The commutator is $(a,b):=aba^{-1}b^{-1}$. The set $\cR\subseteq F(\cX)$ and the auxiliary elements $\wt_{-1}(s)$, $\wt_{\ell,jk}(s)$ are given as follows, for all $s,t\in\Cx$, $u,v\in\mathbb{C}$ and $(\ell,j,k),(m,p,q)\in E$.
\begingroup\allowdisplaybreaks
\begin{align*}
\tag{Re:0}
  \wt_{-1}(s) &:= X_{-1}(s)\,Y_{-1}(-s^{-1})\,X_{-1}(s), \qquad \wt_{-1}:=\wt_{-1}(1),\\
\tag{Im:0}
  \wt_{\ell,jk}(s) &:= X_{\ell,jk}(s)\,Y_{\ell,jk}\Bigl(\frac{-s^{-1}}{c_{\ell j}}\Bigr)\,X_{\ell,jk}(s), \qquad \wt_{\ell,jk}:=\wt_{\ell,jk}(1),\\
\intertext{Relations defining a subgroup $\mathrm{GL}_2(-1)$:}
\tag{H:1a} H_1(s)H_1(t)&=H_1(st),\\
\tag{H:1b} H_2(s)H_2(t)&=H_2(st),\\
\tag{H:2} H_1(s)H_2(t)&=H_2(t)H_1(s),\\
\tag{Re:1a} X_{-1}(u)X_{-1}(v)&=X_{-1}(u+v),\\
\tag{Re:1b} Y_{-1}(u)Y_{-1}(v)&=Y_{-1}(u+v),\\
\tag{Re:2} Y_{-1}(-t)X_{-1}(s)Y_{-1}(t)&=X_{-1}(-t^{-1})\,Y_{-1}(-t^{2}s)\,X_{-1}(t^{-1}),\\
\tag{Re:3} \wt_{-1}(s)\wt_{-1}&=H_1(-s)H_2(-s^{-1}),\\
\tag{Re:4a} \wt_{-1}X_{-1}(u)\wt_{-1}^{-1}&=Y_{-1}(-u),\\
\tag{Re:4b} \wt_{-1}Y_{-1}(u)\wt_{-1}^{-1}&=X_{-1}(-u),\\
\tag{Re:5a} \wt_{-1}H_1(s)\wt_{-1}^{-1}&=H_2(s),\\
\tag{Re:5b} \wt_{-1}H_2(s)\wt_{-1}^{-1}&=H_1(s),\\
\tag{Re:6a} H_1(s)X_{-1}(u)H_1(s)^{-1}&=X_{-1}(su),\\
\tag{Re:6b} H_2(s)X_{-1}(u)H_2(s)^{-1}&=X_{-1}(s^{-1}u),\\
\tag{Re:6c} H_1(s)Y_{-1}(u)H_1(s)^{-1}&=Y_{-1}(s^{-1}u),\\
\tag{Re:6d} H_2(s)Y_{-1}(u)H_2(s)^{-1}&=Y_{-1}(su),\\
\intertext{Relations defining subgroups $\mathrm{GL}_2(\ell,j,k)$:}
\tag{Im:1a} X_{\ell,jk}(u)X_{\ell,jk}(v)&=X_{\ell,jk}(u+v),\\
\tag{Im:1b} Y_{\ell,jk}(u)Y_{\ell,jk}(v)&=Y_{\ell,jk}(u+v),\\
\tag{Im:2} Y_{\ell,jk}(-t)X_{\ell,jk}(s)Y_{\ell,jk}(t)&=X_{\ell,jk}\Bigl(\frac{-t^{-1}}{c_{\ell j}}\Bigr)\,Y_{\ell,jk}(-c_{\ell j}t^2s)\,X_{\ell,jk}\Bigl(\frac{t^{-1}}{c_{\ell j}}\Bigr),\\
\tag{Im:3} \wt_{\ell,jk}\bigl(s^{(\ell+1)(j-\ell)}\bigr)\wt_{\ell,jk}&=H_1\bigl((-s)^{j-\ell}\bigr)H_2\bigl((-s)^{\ell+1}\bigr),\\
\tag{Im:4a} \wt_{\ell,jk}X_{\ell,jk}(u)\wt_{\ell,jk}^{-1}&=Y_{\ell,jk}\Bigl(\frac{-u}{c_{\ell j}}\Bigr),\\
\tag{Im:4b} \wt_{\ell,jk}Y_{\ell,jk}(u)\wt_{\ell,jk}^{-1}&=X_{\ell,jk}(-c_{\ell j}u),\\
\tag{Im:5a} \wt_{\ell,jk}H_1\bigl(s^{j-\ell}\bigr)\wt_{\ell,jk}^{-1}&=H_2\bigl(s^{-(\ell+1)}\bigr),\\
\tag{Im:5b} \wt_{\ell,jk}H_2\bigl(s^{\ell+1}\bigr)\wt_{\ell,jk}^{-1}&=H_1\bigl(s^{-(j-\ell)}\bigr),\\
\tag{Im:6a} H_1(s)X_{\ell,jk}(u)H_1(s)^{-1}&=X_{\ell,jk}(s^{\ell+1}u),\\
\tag{Im:6b} H_2(s)X_{\ell,jk}(u)H_2(s)^{-1}&=X_{\ell,jk}(s^{j-\ell}u),\\
\tag{Im:6c} H_1(s)Y_{\ell,jk}(u)H_1(s)^{-1}&=Y_{\ell,jk}(s^{-(\ell+1)}u),\\
\tag{Im:6d} H_2(s)Y_{\ell,jk}(u)H_2(s)^{-1}&=Y_{\ell,jk}(s^{-(j-\ell)}u),\\
\intertext{Relations between the unipotent generators:}
\tag{U:1a} (X_{-1}(u),X_{j-1,jk}(v))&=1,\\
\tag{U:1b} (Y_{-1}(u),Y_{j-1,jk}(v))&=1,\\
\tag{U:1c} (Y_{-1}(u),X_{0,jk}(v))&=1,\\
\tag{U:1d} (X_{-1}(u),Y_{0,jk}(v))&=1,\\
\tag{U:2} (X_{\ell,jk}(u),Y_{m,pq}(v))&=1\quad\text{for } j\neq p,\text{ or } k\neq q,\text{ or } |\ell-m|>1,\\
\tag{U:3a} (X_{1,2k}(u),Y_{0,2k}(v))&=X_{-1}(uv),\\
\tag{U:3b} (X_{0,2k}(u),Y_{1,2k}(v))&=Y_{-1}(uv),\\
\intertext{Action of $\wt_{-1}$ on the unipotent generators:}
\tag{U:4a} \wt_{-1}X_{\ell,jk}(u)\wt_{-1}^{-1}&=X_{j-1-\ell,jk}\bigl((-1)^{j-\ell-1}u\bigr),\\
\tag{U:4b} \wt_{-1}Y_{\ell,jk}(u)\wt_{-1}^{-1}&=Y_{j-1-\ell,jk}\bigl((-1)^{j-\ell-1}u\bigr).
\end{align*}
\endgroup

\paragraph{The group.} The group is
\[
G(\mathfrak{m})=\langle\cX\mid\cR\rangle=F(\cX)/N_\cR,
\]
where $N_\cR$ is the normal closure of $\cR$.

\begin{theorem}[Corollary 3.2]
For all $u\in\mathbb{C}$, $X_{-1}(u)^{-1}=X_{-1}(-u)$ in $G(\mathfrak{m})$.
\end{theorem}

\[
\ell<j \quad\text{and}\quad 1\leq k\leq c(j).
\]
This is the source's $E$ with $c$ in place of the coefficients of $J$.

\paragraph{The generating symbols.} The set of symbols $\cX_c$ is the disjoint union of the following families:
\begin{itemize}

\paragraph{The constants $c_{\ell j}$.} For $e=(\ell,j,k)\in E_c$,
\[
c_e:=(-1)^{\ell+1}\binom{j-1}{\ell}(\ell+1)(j-\ell)\in\mathbb{C}.
\]
The differences $j-1$ and $j-\ell$ are computed in $\mathbb{N}$ by truncated subtraction. Since $\ell<j$ forces $j\geq 1$, both are the true differences. Hence $c_e=c_{\ell j}$.

\paragraph{The relators.} The set $\cR_c\subseteq F(\cX_c)$ consists of the words $L\,R^{-1}$, one for each source relation $L=R$. For relations of the form $(a,b)=1$ the word is the commutator $(a,b)$ itself. The families are listed below. Here $s,t$ range over $\Cx$, $u,v$ over $\mathbb{C}$, and $e,f,e'$ over $E_c$. Powers of $s$ with exponent in $\mathbb{N}$ or $\mathbb{Z}$ are taken in $\Cx$ when they are arguments of $H_i$ or of $\wt_e$, and in $\mathbb{C}$ when they are coefficients inside the arguments of $X_e$ and $Y_e$. Abbreviate $\ell=\ell_e$, $j=j_e$, $k=k_e$.
\begin{itemize}
\item (H:1a), (H:1b): $H_1(s)H_1(t)H_1(st)^{-1}$ and $H_2(s)H_2(t)H_2(st)^{-1}$.
\item (H:2): $H_1(s)H_2(t)\bigl(H_2(t)H_1(s)\bigr)^{-1}$.
\item (Re:1a), (Re:1b): $X_{-1}(u)X_{-1}(v)X_{-1}(u+v)^{-1}$ and $Y_{-1}(u)Y_{-1}(v)Y_{-1}(u+v)^{-1}$.
\item (Re:2): $Y_{-1}(-t)X_{-1}(s)Y_{-1}(t)\bigl(X_{-1}(-t^{-1})Y_{-1}(-t^2s)X_{-1}(t^{-1})\bigr)^{-1}$.
\item (Re:3): $\wt_{-1}(s)\wt_{-1}(1)\bigl(H_1(-s)H_2(-s^{-1})\bigr)^{-1}$.
\item (Re:4a)--(Re:5b): $\wt_{-1}(1)\,g\,\wt_{-1}(1)^{-1}\,h^{-1}$ for the pairs $(g,h)$ equal to $(X_{-1}(u),Y_{-1}(-u))$, $(Y_{-1}(u),X_{-1}(-u))$, $(H_1(s),H_2(s))$ and $(H_2(s),H_1(s))$.
\item (Re:6a)--(Re:6d): $H_1(s)X_{-1}(u)H_1(s)^{-1}X_{-1}(su)^{-1}$, together with the analogous words for $X_{-1}(s^{-1}u)$ under $H_2$, for $Y_{-1}(s^{-1}u)$ under $H_1$, and for $Y_{-1}(su)$ under $H_2$.
\item (Im:1a), (Im:1b): $X_e(u)X_e(v)X_e(u+v)^{-1}$ and $Y_e(u)Y_e(v)Y_e(u+v)^{-1}$.
\item (Im:2): $Y_e(-t)X_e(s)Y_e(t)\bigl(X_e(-t^{-1}/c_e)\,Y_e(-c_et^2s)\,X_e(t^{-1}/c_e)\bigr)^{-1}$.

\section{Proof}

\subsection{Overview}

The source proof has two steps. It reads (Re:1a) as $X_{-1}(u)X_{-1}(v)=X_{-1}(u+v)$ in $G(\mathfrak{m})$, and it sets $v=-u$ and uses $X_{-1}(0)=1$, which it takes from Lemma 3.1. The formal proof follows the same two steps.
\begin{itemize}
\item \cref{lem:Xm_add_in_G} is the reading of (Re:1a) as an identity in $G_c$.
\item \cref{lem:Xm_zero_in_G} is the fact $X_{-1}(0)=1$.
\end{itemize}
The only deviation is that Lemma 3.1 of the source is not imported. The fact $X_{-1}(0)=1$ is instead derived directly from the instance $u=v=0$ of (Re:1a). \Cref{lem:mk_rel_eq_one_in_G} records the general fact that every relator becomes trivial in $G_c$. It is not used by the later declarations, which invoke the same fact directly for the specific relators they need.
